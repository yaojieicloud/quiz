# 用途：将 topics.sort_order 从 INTEGER 改为 REAL（支持浮点插入排序算法）
# 执行：python src/migrations/upgrade_sort_order.py
import sqlite3, sys

DB = sys.argv[1] if len(sys.argv) > 1 else "data/quiz.db"
conn = sqlite3.connect(DB)
cur = conn.cursor()

# SQLite 不支持 ALTER COLUMN，直接重建
# Step 1: 建新表
cur.execute("""
CREATE TABLE topics_new (
    id INTEGER PRIMARY KEY,
    subject_id INTEGER NOT NULL,
    name TEXT NOT NULL,
    unit TEXT,
    sort_order REAL NOT NULL DEFAULT 0,
    deprecated INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (subject_id) REFERENCES subjects(id)
);
""")

# Step 2: 复制数据（INTEGER 自动转 REAL）
cur.execute("INSERT INTO topics_new (id, subject_id, name, unit, sort_order, deprecated, created_at) SELECT id, subject_id, name, unit, sort_order, deprecated, created_at FROM topics;")

# Step 3: 删除旧表
cur.execute("DROP TABLE topics;")

# Step 4: 重命名
cur.execute("ALTER TABLE topics_new RENAME TO topics;")

conn.commit()
conn.close()
print("Done: topics.sort_order → REAL")

## 2. DELETE /api/topics/{id} 改造（软/硬删除分流）

### 2.1 新增 Schema

```python
# src/schemas.py 新增

class DeleteTopicResult(BaseModel):
    """删除课程返回值"""
    mode: Literal["soft", "hard"]
    topic_count: int = 1
    question_count: int = 0
    message: str

class ReorderTopicRequest(BaseModel):
    """拖拽重排请求"""
    id: int                      # 被拖动的课程 ID
    prev_id: int | None          # 前一个课程 ID（None=拖到最前）
    next_id: int | None          # 后一个课程 ID（None=拖到最后）
```

### 2.2 删除逻辑实现

```python
# src/routers/subjects.py

@router.delete("/{topic_id}", response_model=DeleteTopicResult)
async def delete_topic(topic_id: int, db: Session = Depends(get_db)):
    topic = db.query(Topic).filter(Topic.id == topic_id).first()
    if not topic:
        raise HTTPException(404, "课程不存在")

    # 1. 查题目数量
    q_count = db.query(Question).filter(Question.topic_id == topic_id).count()

    # 2. 查有学员做题记录的题目数量（关键：查 answer_records，不要查 exam_records）
    # exam_records 是聚合表（无 question_id），查 answer_records.question_id 才是正确的
    if q_count > 0:
        q_ids = [r.id for r in db.query(Question.id).filter(Question.topic_id == topic_id).all()]
        done_count = db.query(AnswerRecord.id).filter(
            AnswerRecord.question_id.in_(q_ids)
        ).distinct().count()
    else:
        done_count = 0

    # 3. 判断软/硬删
    if q_count > 0 and done_count > 0:
        # 软删
        topic.deprecated = 1
        db.query(Question).filter(Question.topic_id == topic_id).update(
            {Question.deprecated: True}, synchronize_session=False
        )
        db.commit()
        msg = f"课程「{topic.name}」及{q_count}道题目已软删除（不可见，历史记录保留）"
        return DeleteTopicResult(mode="soft", topic_count=1, question_count=q_count, message=msg)
    else:
        # 硬删：级联删题目
        db.query(Question).filter(Question.topic_id == topic_id).delete(synchronize_session=False)
        db.delete(topic)
        db.commit()
        msg = f"课程「{topic.name}」及{q_count}道题目已永久删除"
        return DeleteTopicResult(mode="hard", topic_count=1, question_count=q_count, message=msg)
```

> ⚠️ **数据安全红线**：必须查 `answer_records.question_id`。`exam_records` 表无 question_id 列，若查错会得到 done_count=0，**所有有历史数据的课程都会被硬删，数据永久丢失**。

## 3. PUT /api/topics/reorder 新增

```python
# src/routers/topics.py

@router.put("/reorder")
async def reorder_topic(body: ReorderTopicRequest, db: Session = Depends(get_db)):
    topic = db.query(Topic).filter(Topic.id == body.id).first()
    if not topic:
        raise HTTPException(404, "课程不存在")

    prev_sort = None
    next_sort = None

    if body.prev_id is not None:
        prev = db.query(Topic).filter(Topic.id == body.prev_id).first()
        if prev:
            prev_sort = prev.sort_order

    if body.next_id is not None:
        nxt = db.query(Topic).filter(Topic.id == body.next_id).first()
        if nxt:
            next_sort = nxt.sort_order

    if prev_sort is None and next_sort is None:
        # 只有一个课程，无需排序
        return {"sort_order": topic.sort_order}

    if prev_sort is None:
        # 拖到最前：new = next / 2
        new_sort = next_sort / 2
    elif next_sort is None:
        # 拖到最后：new = prev + 1024
        new_sort = prev_sort + 1024
    else:
        # 两者之间：new = (prev + next) / 2
        new_sort = (prev_sort + next_sort) / 2

    topic.sort_order = new_sort
    db.commit()
    return {"id": topic.id, "sort_order": new_sort}
```
