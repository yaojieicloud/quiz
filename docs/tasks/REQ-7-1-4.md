### 3.1 页面骨架

页面包含：
- 顶栏 `#topbar`（复用 common.js 的顶栏）
- 标题区：返回按钮 + 科目名 + 当前课程名
- Tab 区：「📺 学习」/「📝 练习」两个 Tab
- 学习 Tab 内容：
  - 左侧课节目录（220px），显示该科目所有 topic
  - 右侧 iframe 区：视频 iframe + 课本 iframe（两个独立区域，URL 为空则隐藏）
  - 底部：「开始做题」按钮（跳 home.html?pick=subjectId）
- 练习 Tab 内容：占位文字「点击下方按钮开始组题」+「开始组题」按钮（跳 home.html?pick=subjectId）

### 3.2 关键代码片段

```javascript
// 从 URL 取 subject_id（默认第一课）
const params = new URLSearchParams(location.search);
const subjectId = parseInt(params.get('subject_id'));
let topics = [];
let currentTopicIdx = 0;

async function init() {
  const sub = (await API.get('/api/subjects')).find(s => s.id === subjectId);
  document.getElementById('subjName').textContent = sub.icon + ' ' + sub.name;
  topics = await API.get(`/api/subjects/${subjectId}/topics`);
  renderTopicList();
  // 默认选第一课
  selectTopic(0);
}

function renderTopicList() {
  const html = topics.map((t, i) => `
    <div class="study-topic-row ${i === currentTopicIdx ? 'active' : ''}" 
         data-idx="${i}" onclick="selectTopic(${i})">
      ${t.name}
      ${(t.tutorial_video_url || t.tutorial_book_url) ? '<span class="has-tut">📚</span>' : ''}
    </div>
  `).join('');
  document.getElementById('topicList').innerHTML = html;
}

function selectTopic(idx) {
  currentTopicIdx = idx;
  const t = topics[idx];
  document.getElementById('topicName').textContent = t.name;
  // 视频
  const vBox = document.getElementById('videoBox');
  if (t.tutorial_video_url) {
    vBox.style.display = '';
    document.getElementById('videoFrame').src = t.tutorial_video_url;
  } else {
    vBox.style.display = 'none';
    document.getElementById('videoFrame').src = 'about:blank';
  }
  // 课本
  const bBox = document.getElementById('bookBox');
  if (t.tutorial_book_url) {
    bBox.style.display = '';
    document.getElementById('bookFrame').src = t.tutorial_book_url;
  } else {
    bBox.style.display = 'none';
    document.getElementById('bookFrame').src = 'about:blank';
  }
  // 降级
  const empty = document.getElementById('emptyHint');
  empty.style.display = (!t.tutorial_video_url && !t.tutorial_book_url) ? '' : 'none';
  renderTopicList();
  // 上一课/下一课按钮
  document.getElementById('prevBtn').disabled = idx === 0;
  document.getElementById('nextBtn').disabled = idx === topics.length - 1;
}

function goPrev() { if (currentTopicIdx > 0) selectTopic(currentTopicIdx - 1); }
function goNext() { if (currentTopicIdx < topics.length - 1) selectTopic(currentTopicIdx + 1); }

function goHome() { location.href = 'home.html'; }
function startQuiz() { location.href = `home.html?pick=${subjectId}`; }

function switchTab(tab) {
  document.querySelectorAll('.study-tab').forEach(el => el.classList.remove('active'));
  document.querySelectorAll('.study-tab-content').forEach(el => el.classList.remove('active'));
  if (tab === 'learn') {
    document.getElementById('tabLearn').classList.add('active');
    document.getElementById('tabLearnContent').classList.add('active');
  } else {
    document.getElementById('tabQuiz').classList.add('active');
    document.getElementById('tabQuizContent').classList.add('active');
  }
}
```
