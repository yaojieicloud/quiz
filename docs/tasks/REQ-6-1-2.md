### 2.3 初始化 Sortable

```js
// 在 toggleTopics() 渲染完成后：
const listEl = el.querySelector('.topics-list');
if (listEl && typeof Sortable !== 'undefined') {
  Sortable.create(listEl, {
    animation: 150,
    handle: '.drag-handle',
    onEnd: async (evt) => {
      const movedId = parseInt(evt.item.dataset.topicId);
      const prevEl = evt.item.previousElementSibling;
      const nextEl = evt.item.nextElementSibling;
      const prevId = prevEl ? parseInt(prevEl.dataset.topicId) : null;
      const nextId = nextEl ? parseInt(nextEl.dataset.topicId) : null;

      try {
        await API.put('/api/topics/reorder', { id: movedId, prev_id: prevId, next_id: nextId });
        toast('已重新排序', 'success');
      } catch(e) {
        toast('排序失败: ' + e.message, 'error');
        // 失败回滚 - 重新加载
        toggleTopics(subjectId);
      }
    }
  });
}
```

注意：每行课程必须是 `<div class="topic-row" data-topic-id="X">` 才能被 Sortable 识别。

## 3. 删除确认弹窗改造

```js
async function confirmDeleteTopic(topicId, subjectId) {
  // 1. 预查询：调用轻量接口或用现有 cache
  const topic = (topicsCache[subjectId] || []).find(t => t.id === topicId);
  if (!topic) {
    toast('课程信息未找到', 'error');
    return;
  }

  // 2. 调用新接口（删除前先 GET 影响预估，或直接 DELETE 走软/硬分支）
  // 方案：直接 DELETE，从返回值判断软/硬
  showConfirm(
    '删除课程',
    `确定删除「${esc(topic.name)}」？<br>${topic.question_count > 0 ? `该课程下有 ${topic.question_count} 道题目。` : ''}`,
    async () => {
      try {
        const res = await API.del('/api/topics/' + topicId);
        if (res.mode === 'soft') {
          toast(res.message, 'success');
        } else {
          toast(res.message, 'success');
        }
        toggleTopics(subjectId);  // 重新加载展开区
        // 刷新 subjectCache（更新题数）
        const subs = await API.get('/api/subjects');
        // 更新全局 subjects 引用
      } catch(e) {
        toast(e.message, 'error');
      }
    }
  );
}
```

**增强版**：删除前先请求影响预估，但需新增 `GET /api/topics/{id}/delete-preview` 接口。本期先直接 DELETE 走分支。

> 📌 **2026-09-01 补充 BUG-4**：删除弹窗 showConfirm 的 msg 用了 HTML 标签（`<b>` `<br>`）但渲染走 `textContent`，导致 HTML 标签原样显示。
